# Dev / editable-local install of the AgentShip framework.
#
# The framework is not on PyPI yet, so the pinned `agentship[starter]==0.0.1` in
# pyproject.toml cannot resolve from an index. Until it publishes, install the three
# framework packages editable from the sibling monorepo checkout, plus the test
# tooling. Run this from the demo repo root, inside a venv:
#
#     python -m venv .venv && source .venv/bin/activate
#     pip install -r requirements-dev.txt
#
# (or just `make install`, which does exactly this.)

-e ../agentship/packages/agentship-core
-e ../agentship/packages/agentship-langgraph
-e ../agentship/packages/agentship-cli
# Phase 07 observability slice: the OTel span-tree + exporters (console/Opik/LangFuse/LangSmith).
# The pinned form is `agentship[starter,observability]`; editable-local it's this sibling package.
-e ../agentship/packages/agentship-observability

pytest>=8
pytest-asyncio>=0.24

# Read-back for the live observability slice: query each hosted backend's API to confirm the
# trace landed. httpx (Opik/LangFuse REST) is already a framework transitive dep; langsmith is
# the LangSmith SDK the read-back uses.
langsmith>=0.1

# The generic chat UI (demos/chat_ui.py, `make ui`) — a browser chat to drive any agent,
# including the deep-research agent's pause→resume, without a per-demo script.
#
# Note: gradio caps pydantic (<=2.12.3) below the floor a framework transitive dep
# (google-genai) prefers (>=2.12.5), so `pip` prints a harmless conflict warning and
# settles on 2.12.3. The demo's LangGraph→LiteLLM path never touches google-genai, so
# every slice runs green at that version. If you don't want the UI, this line is the
# only thing to drop.
gradio>=4

# Firecrawl backs the built-in web_search + scrape_url tools when FIRECRAWL_API_KEY is set (free
# tier at https://firecrawl.dev). It's what lets the deep-research agent actually search the web
# and read full pages. Optional: without it (and without a BRAVE_API_KEY) those tools return an
# actionable setup message and agents answer from model knowledge.
firecrawl-py>=4
